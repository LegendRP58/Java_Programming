data segment para
    scan_table  db 10h, 11h, 12h, 13h, 14h, 15h, 16h, 17h, 18h, 19h
                db 1Eh, 1Fh, 20h, 21h, 22h, 23h, 24h, 25h, 26h
                db 2Ch, 2Dh, 2Eh, 2Fh, 30h, 31h, 32h

    freq_table  dw 262, 277, 294, 311, 330, 349, 370, 392, 415, 440
                dw 466, 494, 523, 554, 587, 622, 659, 698, 740
                dw 784, 831, 880, 932, 988, 1047, 1109

    TABLE_SIZE  equ 26
    BASE_FREQ   dd 1193180

    current_freq    dw 0
    current_div     dw 0
    last_scan       db 0
    start_time      dd 0        ; Время начала (через INT 1Ah)
    sound_on        db 0
    note_duration   dd 5000     ; Длительность в микросекундах (~0.5 сек)

    msg_help    db 'PIANINO MODE.', 0Dh, 0Ah
                db 'Rows: Q-P, A-L, Z-M', 0Dh, 0Ah
                db 'ESC - exit.', 0Dh, 0Ah, '$'
data ends

code segment para
piano_lab proc far
assume cs:code, ds:data, ss:nothing
    push ds
    mov ax, 0
    push ax
    mov ax, data
    mov ds, ax

    lea dx, msg_help
    mov ah, 09h
    int 21h

main_loop:
    ; Проверяем таймер
    call check_timer_proc

    ; Проверяем клавиатуру
    mov ah, 01h
    int 16h
    je no_key
    
    call process_key_proc

no_key:
    jmp main_loop

piano_lab endp

process_key_proc proc near
    push ax
    push bx
    push cx
    push si
    push di
    push dx

    mov ah, 00h
    int 16h

    cmp al, 1Bh
    jne search_key
    jmp exit_prog

search_key:
    xor di, di
    mov bl, ah
    mov cx, TABLE_SIZE
    lea si, scan_table

find_loop:
    mov al, [si]
    cmp al, bl
    je found
    inc si
    inc di
    loop find_loop
    jmp end_key_proc

found:
    cmp ah, last_scan
    jne new_key
    
    cmp sound_on, 1
    je end_key_proc

new_key:
    mov last_scan, ah
    
    mov bx, di
    shl bx, 1
    lea si, freq_table
    mov ax, [si+bx]
    cmp ax, 0
    je end_key_proc

    mov current_freq, ax

    mov dx, word ptr BASE_FREQ+2
    mov ax, word ptr BASE_FREQ
    mov bx, current_freq
    div bx
    mov current_div, ax

    call set_timer_freq
    mov sound_on, 1

    ; Получаем текущее время через INT 1Ah
    mov ah, 00h
    int 1Ah
    ; CX:DX = количество тиков с полуночи
    mov word ptr start_time, dx
    mov word ptr start_time+2, cx

end_key_proc:
    pop dx
    pop di
    pop si
    pop cx
    pop bx
    pop ax
    ret
process_key_proc endp

check_timer_proc proc near
    push ax
    push bx
    push cx
    push dx

    cmp sound_on, 1
    jne end_check

    ; Получаем текущее время
    mov ah, 00h
    int 1Ah
    mov bx, dx          ; BX = текущие тики (младшее слово)
    mov cx, word ptr start_time
    mov dx, word ptr start_time+2
    
    ; Вычисляем разницу
    sub bx, cx          ; BX = прошло тиков
    jns positive
    add bx, 0           ; Обработка переполнения
positive:
    
    ; 18.2 тика в секунду, нужно ~9 тиков = 0.5 сек
    cmp bx, 9
    jl end_check

    call stop_sound
    mov sound_on, 0
    mov last_scan, 0

end_check:
    pop dx
    pop cx
    pop bx
    pop ax
    ret
check_timer_proc endp

set_timer_freq proc near
    push ax

    mov al, 0B6h
    out 43h, al

    mov ax, current_div
    out 42h, al
    mov al, ah
    out 42h, al

    in al, 61h
    or al, 03h
    out 61h, al

    pop ax
    ret
set_timer_freq endp

stop_sound proc near
    push ax
    in al, 61h
    and al, 0FCh
    out 61h, al
    pop ax
    ret
stop_sound endp

exit_prog:
    call stop_sound
    mov ah, 4Ch
    int 21h

code ends
end piano_lab
